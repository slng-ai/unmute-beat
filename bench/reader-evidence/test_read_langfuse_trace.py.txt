"""Offline reader checks: python3 -m unittest discover -s scripts -p test_read_langfuse_trace.py."""

import contextlib
import io
import unittest

import read_langfuse_trace as reader


class ReaderTest(unittest.TestCase):
    def test_n4_aliases_and_missing_values(self):
        # Retained N4 observation 56b92da4bdbe2af8; aliases are not extra tokens.
        usage = reader.usage_of(
            {
                "usageDetails": {
                    "cached_content_tokens": 3480,
                    "text_prompt_tokens": 3907,
                    "text_cache_tokens": 3480,
                    "text_candidates_tokens": 19,
                    "input": 427,
                    "output": 19,
                    "input_cached_tokens": 3480,
                    "total": 14812,
                }
            }
        )
        self.assertEqual(
            usage,
            {
                "input": 3907,
                "output": 19,
                "cached": 3480,
                "total": 3926,
                "unit": "tokens",
            },
        )
        self.assertEqual(
            reader.usage_of(
                {
                    "usageDetails": {
                        "input": 427,
                        "output": 19,
                        "input_cached_tokens": 3480,
                    }
                }
            ),
            usage,
        )
        self.assertIsNone(reader.usage_of({"usageDetails": {"total": 123}})["total"])
        self.assertEqual(
            reader.usage_of({"usageDetails": {"input": 0, "output": 0}})["total"], 0
        )
        self.assertIsNone(
            reader.usage_of({"usageDetails": {"input": True, "output": 2}})["total"]
        )
        self.assertEqual(
            reader.usage_of(
                {
                    "usageDetails": {
                        "text_prompt_tokens": 0,
                        "input": 20,
                        "output": 0,
                    }
                }
            )["input"],
            0,
        )

    def test_speech_units_and_duplicate_observation(self):
        span = {
            "traceId": "t",
            "id": "r",
            "type": "GENERATION",
            "name": "llm",
            "usageDetails": {"input": 10, "output": 2},
        }
        speech = {
            "type": "GENERATION",
            "name": "tts",
            "usageDetails": {"characters": 42},
        }
        with contextlib.redirect_stdout(io.StringIO()) as stream:
            reader.print_usage([span, span, speech])
        output = stream.getvalue()
        llm = next(
            line.split()
            for line in output.splitlines()
            if line.strip().startswith("llm")
        )
        self.assertEqual(llm, ["llm", "1", "10", "2", "-", "12", "12", "12", "tokens"])
        tts = next(
            line.split()
            for line in output.splitlines()
            if line.strip().startswith("tts")
        )
        self.assertEqual(
            tts, ["tts", "1", "42", "-", "-", "42", "42", "42", "characters"]
        )

    def test_median_and_nearest_rank_tail(self):
        spans: list[dict] = [{"name": "llm", "latency": value} for value in (1, 2, 3, 4, 5, 6)]
        spans += [
            {"name": "llm", "latency": value} for value in (-1, float("nan"), True)
        ]
        with contextlib.redirect_stdout(io.StringIO()) as stream:
            reader.print_latency(spans)
        output = stream.getvalue()
        self.assertIn("p50: median; p90: nearest rank", output)
        row = next(
            line.split()
            for line in output.splitlines()
            if line.strip().startswith("llm")
        )
        self.assertEqual(row, ["llm", "6", "3.500", "3.500", "6.000", "6.000"])


if __name__ == "__main__":
    unittest.main()
